/* ========== 插入点 0：原第 22 行 require_once __DIR__ . '/lib.php'; 之后 ========== */

/* 手写字词库：可选依赖，缺失或加载失败时静默降级为原识别逻辑 */
$bglxCandidates = array(
    dirname(__DIR__) . '/handwriting-lexicon/lib/lexicon.php',
    dirname(__DIR__, 2) . '/public/plugins/handwriting-lexicon/lib/lexicon.php',
);
foreach ($bglxCandidates as $bglxFile) {
    if (is_file($bglxFile)) {
        try {
            require_once $bglxFile;
        } catch (Throwable $bglxE) {
            error_log('[handwriting-lexicon] 引入失败：' . $bglxE->getMessage());
        }
        break;
    }
}
$GLOBALS['BG_BGLX_WARNINGS'] = array();

/**
 * 收集词库 suggest 档提示，最后统一并入 $warnings（不改动任何函数签名）。
 *
 * @param array $r 词库查询结果
 * @param string $fieldName 字段名，用于提示文案
 */
function bg_baidu_lexicon_collect($r, $fieldName)
{
    if (!isset($r['action']) || $r['action'] !== 'suggest') {
        return;
    }
    $text = $fieldName . '「' . $r['original'] . '」疑似「' . $r['corrected'] . '」，请人工确认';
    if (!in_array($text, $GLOBALS['BG_BGLX_WARNINGS'], true)) {
        $GLOBALS['BG_BGLX_WARNINGS'][] = $text;
    }
}

/* ========== 插入点 ①：bg_baidu_norm() 函数定义之后（原第 131~136 行后） ========== */

/**
 * 图号两级纠错（字符形近归一 → 词条匹配）。
 * 未命中 / 词库不可用 / 非 auto 时原样返回。
 *
 * @param string $raw 原始图号或包含图号的文本
 * @return string
 */
function bg_baidu_lexicon_partno($raw)
{
    $raw = (string)$raw;
    if ($raw === '' || !function_exists('bglx_correct_partno')) {
        return $raw;
    }
    try {
        $r = bglx_correct_partno($raw, array('plugin' => 'baidu-ocr'));
        if (isset($r['action']) && $r['action'] === 'auto') {
            return (string)$r['corrected'];
        }
        bg_baidu_lexicon_collect($r, '图号');
        return $raw;
    } catch (Throwable $bglxE) {
        return $raw;
    }
}

/**
 * 工序词纠错（process_synonym）。suggest 档保留原值并收集提示。
 *
 * @param string $raw 原始工序词
 * @return string
 */
function bg_baidu_lexicon_process($raw)
{
    $raw = (string)$raw;
    if ($raw === '' || !function_exists('bglx_norm_process')) {
        return $raw;
    }
    try {
        $r = bglx_norm_process($raw, array('plugin' => 'baidu-ocr'));
        if (isset($r['action']) && $r['action'] === 'auto') {
            return (string)$r['corrected'];
        }
        bg_baidu_lexicon_collect($r, '工序');
        return $raw;
    } catch (Throwable $bglxE) {
        return $raw;
    }
}

/**
 * 单位词纠错（unit_synonym）。
 *
 * @param string $unit 原始单位
 * @return string
 */
function bg_baidu_lexicon_unit($unit)
{
    $unit = (string)$unit;
    if ($unit === '' || !function_exists('bglx_norm_unit')) {
        return $unit;
    }
    try {
        $r = bglx_norm_unit($unit, array('plugin' => 'baidu-ocr'));
        if (isset($r['action']) && $r['action'] === 'auto') {
            return (string)$r['corrected'];
        }
        return $unit; // 单位多为 suggest 档，不提示以免噪声
    } catch (Throwable $bglxE) {
        return $unit;
    }
}
